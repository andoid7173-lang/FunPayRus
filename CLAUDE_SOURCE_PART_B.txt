"""Read-only experimental FunPay sign-in check.

Only GET https://funpay.com/; never follows redirects with credentials,
never sends the golden_key to Telegram, doesn't automate transactions.
FunPay does not provide a stable public API: parsing may stop working.
"""
import html
from html.parser import HTMLParser
import json
import os
import re
import ssl
import time
from urllib import error, request

_KEY = re.compile(r"^[A-Za-z0-9_-]{16,256}$")
_MAX_HTML_BYTES = 2 * 1024 * 1024
_CACHE = {"until": 0.0, "response": ""}


class ProbeError(Exception):
    """Known, deliberately non-sensitive failure message."""


class _NoRedirect(request.HTTPRedirectHandler):
    def redirect_request(self, req, fp, code, msg, headers, newurl):
        return None


class _ProfileHTML(HTMLParser):
    def __init__(self):
        super().__init__(convert_charrefs=True)
        self.in_user_name = False
        self.username = ""
        self.app_data = None

    def handle_starttag(self, tag, attrs):
        attrs = dict(attrs)
        if tag == "body":
            self.app_data = attrs.get("data-app-data")
        if tag == "div" and "user-link-name" in attrs.get("class", "").split():
            self.in_user_name = True

    def handle_endtag(self, tag):
        if tag == "div":
            self.in_user_name = False

    def handle_data(self, data):
        if self.in_user_name and not self.username:
            self.username = data.strip()[:60]


def verify(golden_key, opener=None, timeout=10, user_agent=None):
    """Returns a short success string or raises a safe ProbeError.

    An injected opener makes this fully testable with no external requests.
    """
    if not isinstance(golden_key, str) or not _KEY.fullmatch(golden_key):
        raise ProbeError("Golden Key не настроен или имеет неверный формат")
    if user_agent is not None and (not isinstance(user_agent, str) or
                                   len(user_agent) > 300 or any(c in user_agent for c in "\r\n")):
        raise ProbeError("Некорректный User-Agent")
    headers = {
        "Cookie": "golden_key=" + golden_key,
        "User-Agent": user_agent or "Mozilla/5.0 FunPayRus/0.3.0b1 (read-only test)",
        "Accept": "text/html",
        "Cache-Control": "no-store",
    }
    req = request.Request("https://funpay.com/", headers=headers, method="GET")
    client = opener if opener is not None else request.build_opener(request.ProxyHandler({}), _NoRedirect())
    try:
        with client.open(req, timeout=timeout) as resp:
            if resp.geturl() != "https://funpay.com/":
                raise ProbeError("FunPay перенаправил запрос; проверка остановлена")
            data = resp.read(_MAX_HTML_BYTES + 1)
            if len(data) > _MAX_HTML_BYTES:
                raise ProbeError("Слишком большой ответ FunPay")
    except ProbeError:
        raise
    except error.HTTPError as exc:
        if exc.code in (301, 302, 303, 307, 308):
            raise ProbeError("FunPay перенаправил запрос; проверка остановлена") from None
        if exc.code in (401, 403):
            raise ProbeError("FunPay отклонил авторизацию") from None
        if exc.code == 429:
            raise ProbeError("FunPay ограничил частоту запросов; повторите позже") from None
        raise ProbeError("FunPay вернул ошибку HTTP") from None
    except (error.URLError, TimeoutError, OSError, ssl.SSLError):
        raise ProbeError("Нет защищённого соединения с FunPay") from None
    try:
        parser = _ProfileHTML()
        parser.feed(data.decode("utf-8"))
        appdata = json.loads(html.unescape(parser.app_data or ""))
        uid = appdata.get("userId") if isinstance(appdata, dict) else None
        if not parser.username or type(uid) is not int or uid <= 0:
            raise ValueError("missing authorized profile")
        # Strip control characters so untrusted markup cannot format Telegram replies.
        username = "".join(c for c in parser.username if c.isprintable())[:60]
        if not username:
            raise ValueError("bad username")
        return "✅ FunPay: авторизация подтверждена (тест)\n👤 Аккаунт: " + username
    except (ValueError, TypeError, UnicodeError):
        raise ProbeError("Авторизация не подтверждена: изменился ответ сайта или ключ недействителен") from None


def status_from_environment(env=None, opener=None):
    """Never expose the key or raw upstream exception text."""
    env = os.environ if env is None else env
    key = env.get("FUNPAY_GOLDEN_KEY", "")
    if not key:
        return ("🔑 FunPay пока не подключён.\n"
                "Настройте Golden Key локально на VPS командой:\n"
                "sudo bash scripts/configure_funpay.sh\n"
                "Не отправляйте ключ в Telegram.")
    if opener is None and time.monotonic() < _CACHE["until"]:
        return _CACHE["response"]
    try:
        result = verify(key, opener=opener, user_agent=env.get("FUNPAY_USER_AGENT"))
    except ProbeError as exc:
        result = "⚠️ FunPay: " + str(exc)
    if opener is None:
        _CACHE["response"] = result  # cache contains no Golden Key
        _CACHE["until"] = time.monotonic() + 60
    return result

"""Read-only, opt-in FunPay sales monitor.

Requests only the current seller's first /orders/trade page. No API writes,
chat messages, credential delivery or login-code processing. HTML may change.
"""
from dataclasses import dataclass
from contextlib import contextmanager
from html.parser import HTMLParser
import html
import json
import os
from pathlib import Path
import re
import sqlite3
import time
from urllib import request, error

from bot.funpay_probe import _KEY, _NoRedirect

ORDERS_URL = 'https://funpay.com/orders/trade'
_MAX_BYTES = 2 * 1024 * 1024
_ORDER_ID = re.compile(r'^[A-Za-z0-9_-]{4,64}$')
VOID = frozenset(('area', 'base', 'br', 'col', 'embed', 'hr', 'img', 'input',
                  'link', 'meta', 'param', 'source', 'track', 'wbr'))


class OrderMonitorError(Exception):
    """Only safe diagnostic errors that never include a Golden Key."""


@dataclass(frozen=True)
class Order:
    order_id: str
    status: str  # paid, closed, refunded
    description: str = ''
    buyer: str = ''
    price: str = ''


def _clean(value, limit=140):
    return ' '.join(''.join(c for c in str(value) if c.isprintable()).split())[:limit]


class _SalesHTML(HTMLParser):
    """Minimal parser for FunPay /orders/trade tc-item cards; strict auth."""
    FIELDS = {'tc-order': 'id', 'order-desc': 'description',
              'tc-price': 'price', 'media-user-name': 'buyer'}

    def __init__(self):
        super().__init__(convert_charrefs=True)
        self.app_data = None
        self.authorized_header = False
        self.orders = []
        self.current = None
        self.depth = 0
        self.active_fields = []

    def handle_starttag(self, tag, attrs):
        attrs = dict(attrs)
        classes = set(attrs.get('class', '').split())
        if tag == 'body':
            self.app_data = attrs.get('data-app-data')
        if 'user-link-name' in classes:
            self.authorized_header = True
        if self.current is None and tag == 'a' and 'tc-item' in classes:
            self.current = {'id': [], 'description': [], 'price': [], 'buyer': [],
                            'status': 'refunded' if 'warning' in classes else
                                      'paid' if 'info' in classes else 'closed'}
            self.depth = 1
            self.active_fields = []
            return
        if self.current is not None:
            if tag not in VOID:
                self.depth += 1
            for css, field in self.FIELDS.items():
                if css in classes and tag in ('div', 'span'):
                    self.active_fields.append((field, self.depth))
                    break

    def handle_startendtag(self, tag, attrs):
        self.handle_starttag(tag, attrs)
        if tag not in VOID:
            self.handle_endtag(tag)

    def handle_data(self, data):
        if self.current is not None:
            for field, _ in self.active_fields:
                self.current[field].append(data)

    def handle_endtag(self, tag):
        if self.current is None or tag in VOID:
            return
        if self.depth == 1 and tag == 'a':
            fields = self.current
            self.current = None
            raw_id = _clean(''.join(fields['id']), 70).lstrip('#')
            if _ORDER_ID.fullmatch(raw_id) and len(self.orders) < 100:
                self.orders.append(Order(
                    raw_id, fields['status'], _clean(''.join(fields['description']), 200),
                    _clean(''.join(fields['buyer']), 80), _clean(''.join(fields['price']), 80)
                ))
            self.depth = 0
            self.active_fields = []
            return
        self.active_fields = [(field, depth) for field, depth in self.active_fields
                              if depth != self.depth]
        self.depth -= 1

    def validate(self):
        try:
            data = json.loads(html.unescape(self.app_data or ''))
            if (not self.authorized_header or not isinstance(data, dict) or
                    type(data.get('userId')) is not int or data['userId'] <= 0):
                raise ValueError('not signed in')
        except (ValueError, TypeError):
            raise OrderMonitorError('Авторизация FunPay не подтверждена; список продаж не прочитан') from None


def fetch_sales(golden_key, opener=None, timeout=12, user_agent=None):
    """Read-only HTTPS GET. No proxy environment, redirects or raw responses in errors."""
    if not isinstance(golden_key, str) or not _KEY.fullmatch(golden_key):
        raise OrderMonitorError('FunPay Golden Key не настроен')
    if user_agent is not None and (not isinstance(user_agent, str) or
                                   len(user_agent) > 300 or any(c in user_agent for c in '\r\n')):
        raise OrderMonitorError('Некорректный User-Agent')
    req = request.Request(ORDERS_URL, headers={
        'Cookie': 'golden_key=' + golden_key,
        'User-Agent': user_agent or 'Mozilla/5.0 FunPayRust/0.3.2b1 (read-only)',
        'Accept': 'text/html', 'Cache-Control': 'no-store',
    }, method='GET')
    client = opener if opener is not None else request.build_opener(request.ProxyHandler({}), _NoRedirect())
    try:
        with client.open(req, timeout=timeout) as resp:
            if resp.geturl() != ORDERS_URL:
                raise OrderMonitorError('Перенаправление FunPay; запрос остановлен')
            data = resp.read(_MAX_BYTES + 1)
        if len(data) > _MAX_BYTES:
            raise OrderMonitorError('Ответ FunPay слишком большой')
    except OrderMonitorError:
        raise
    except error.HTTPError as exc:
        if exc.code in (301, 302, 303, 307, 308):
            raise OrderMonitorError('Перенаправление FunPay; запрос остановлен') from None
        if exc.code in (401, 403):
            raise OrderMonitorError('FunPay отклонил авторизацию') from None
        if exc.code == 429:
            raise OrderMonitorError('FunPay ограничил частоту запросов') from None
        raise OrderMonitorError(f'FunPay HTTP {exc.code}') from None
    except (error.URLError, TimeoutError, OSError):
        raise OrderMonitorError('Нет защищённого соединения с FunPay') from None
    try:
        parser = _SalesHTML()
        parser.feed(data.decode('utf-8'))
        parser.validate()
        return parser.orders
    except (UnicodeError, ValueError) as exc:
        raise OrderMonitorError('Изменился HTML списка продаж FunPay') from None


class OrderStore:
    """Persistent, first-snapshot-silent order storage with safe limited fields."""

    def __init__(self, filename):
        self.filename = Path(filename)
        self.last_error = None  # in-process only; never stores upstream HTML
        try:
            if self.filename.is_symlink() or (self.filename.exists() and
                not self.filename.is_file()):
                raise OrderMonitorError('Небезопасный путь хранения заказов')
            self.filename.parent.mkdir(mode=0o700, parents=True, exist_ok=True)
            with self._connect() as db:
                db.execute('CREATE TABLE IF NOT EXISTS orders ('
                           'id TEXT PRIMARY KEY, status TEXT NOT NULL, '
                           'description TEXT NOT NULL, buyer TEXT NOT NULL, '
                           'price TEXT NOT NULL, updated_at INTEGER NOT NULL)')
                db.execute('CREATE TABLE IF NOT EXISTS metadata ('
                           'name TEXT PRIMARY KEY, value TEXT NOT NULL)')
                # Durable at-least-once Telegram delivery queue. Historical
                # events are queued atomically with the order snapshot.
                db.execute('CREATE TABLE IF NOT EXISTS delivery_outbox ('
                           'id INTEGER PRIMARY KEY AUTOINCREMENT, '
                           'recipient INTEGER NOT NULL, order_id TEXT NOT NULL, '
                           'kind TEXT NOT NULL, previous_status TEXT, '
                           'status TEXT NOT NULL, description TEXT NOT NULL, '
                           'buyer TEXT NOT NULL, price TEXT NOT NULL, '
                           'created_at INTEGER NOT NULL, delivered_at INTEGER)')
                db.execute('CREATE INDEX IF NOT EXISTS delivery_outbox_pending '
                           'ON delivery_outbox(delivered_at, id)')
        except (OSError, sqlite3.Error) as exc:
            raise OrderMonitorError('Не удалось открыть хранилище заказов') from None

    @contextmanager
    def _connect(self):
        if self.filename.is_symlink():
            raise OrderMonitorError('Небезопасный файл заказов')
        db = sqlite3.connect(str(self.filename), timeout=8)
        try:
            os.chmod(self.filename, 0o600)
            with db:
                yield db
        finally:
            db.close()

    def record(self, orders, recipients=()):
        """Store a snapshot and queue alert events in the SAME transaction.

        First successful snapshot is silent, as in v0.3.2. New paid sales and
        later closed/refunded transitions generate events. The return value
        remains a list of newly paid orders for existing callers.
        """
        if not isinstance(orders, list):
            raise OrderMonitorError('Некорректный список заказов')
        recipients = sorted({uid for uid in recipients if type(uid) is int and uid > 0})
        now = int(time.time())
        notifications = []
        try:
            with self._connect() as db:
                db.execute('BEGIN IMMEDIATE')
                first = db.execute("SELECT value FROM metadata WHERE name='initialized'").fetchone() is None
                for order in orders[:100]:
                    if not isinstance(order, Order) or not _ORDER_ID.fullmatch(order.order_id):
                        continue
                    if order.status not in ('paid', 'closed', 'refunded'):
                        continue
                    previous = db.execute(
                        'SELECT status FROM orders WHERE id=?', (order.order_id,)
                    ).fetchone()
                    old_status = previous[0] if previous else None
                    kind = None
                    if not first:
                        if order.status == 'paid' and old_status != 'paid':
                            notifications.append(order)
                            kind = 'paid'
                        elif old_status is not None and old_status != order.status:
                            kind = 'status'
                    description = _clean(order.description, 200)
                    buyer = _clean(order.buyer, 80)
                    price = _clean(order.price, 80)
                    db.execute('INSERT INTO orders(id,status,description,buyer,price,updated_at) '
                               'VALUES (?,?,?,?,?,?) ON CONFLICT(id) DO UPDATE SET '
                               'status=excluded.status, description=excluded.description, '
                               'buyer=excluded.buyer, price=excluded.price, updated_at=excluded.updated_at',
                               (order.order_id, order.status, description, buyer, price, now))
                    if kind:
                        for uid in recipients:
                            db.execute(
                                'INSERT INTO delivery_outbox(recipient,order_id,kind,'
                                'previous_status,status,description,buyer,price,created_at) '
                                'VALUES (?,?,?,?,?,?,?,?,?)',
                                (uid, order.order_id, kind, old_status,
                                 order.status, description, buyer, price, now))
                db.execute("INSERT OR REPLACE INTO metadata(name,value) VALUES('initialized','1')")
                db.execute("INSERT OR REPLACE INTO metadata(name,value) VALUES('last_poll',?)", (str(now),))
            self.last_error = None
            return notifications
        except (sqlite3.Error, OSError):
            raise OrderMonitorError('Не удалось сохранить результаты мониторинга') from None

    def pending_deliveries(self, limit=30):
        """Return (event_id, recipient, kind, previous_status, Order) tuples."""
        if type(limit) is not int or not 1 <= limit <= 100:
            raise ValueError('Invalid delivery limit')
        try:
            with self._connect() as db:
                rows = db.execute(
                    'SELECT id,recipient,kind,previous_status,order_id,status,'
                    'description,buyer,price FROM delivery_outbox '
                    'WHERE delivered_at IS NULL ORDER BY id ASC LIMIT ?', (limit,)
                ).fetchall()
            return [(r[0], r[1], r[2], r[3], Order(*r[4:])) for r in rows]
        except (sqlite3.Error, OSError):
            raise OrderMonitorError('Не удалось прочитать очередь уведомлений') from None

    def mark_delivered(self, event_id):
        try:
            with self._connect() as db:
                db.execute('UPDATE delivery_outbox SET delivered_at=? '
                           'WHERE id=? AND delivered_at IS NULL',
                           (int(time.time()), event_id))
        except (sqlite3.Error, OSError):
            raise OrderMonitorError('Не удалось отметить доставку уведомления') from None

    def find(self, query, limit=7):
        """Search only saved orders, using bound parameters (no wildcard injection)."""
        if not isinstance(query, str) or not 1 <= len(query.strip()) <= 80:
            return []
        q = query.strip()
        try:
            with self._connect() as db:
                rows = db.execute(
                    'SELECT id,status,description,buyer,price FROM orders '
                    'WHERE instr(lower(id),lower(?))>0 OR '
                    'instr(lower(buyer),lower(?))>0 OR '
                    'instr(lower(description),lower(?))>0 '
                    'ORDER BY updated_at DESC, id DESC LIMIT ?',
                    (q, q, q, min(7, max(1, limit)))
                ).fetchall()
            return [Order(*r) for r in rows]
        except (sqlite3.Error, OSError):
            raise OrderMonitorError('Не удалось выполнить поиск заказов') from None

    def get_order(self, order_id):
        if type(order_id) is not str or not _ORDER_ID.fullmatch(order_id):
            return None
        try:
            with self._connect() as db:
                row = db.execute(
                    'SELECT id,status,description,buyer,price FROM orders WHERE id=?',
                    (order_id,)
                ).fetchone()
            return Order(*row) if row else None
        except (sqlite3.Error, OSError):
            raise OrderMonitorError('Не удалось прочитать заказ') from None

    def summary(self):
        try:
            with self._connect() as db:
                count = db.execute('SELECT count(*) FROM orders').fetchone()[0]
                last = db.execute("SELECT value FROM metadata WHERE name='last_poll'").fetchone()
                rows = db.execute('SELECT id,status,description,buyer,price FROM orders '
                                  'ORDER BY updated_at DESC, id DESC LIMIT 7').fetchall()
            return count, int(last[0]) if last else None, [Order(*r) for r in rows]
        except (sqlite3.Error, OSError):
            raise OrderMonitorError('Не удалось прочитать историю заказов') from None


def order_url(order_id):
    """Return a FunPay order permalink for a strictly validated order ID.

    Order IDs come from untrusted HTML. Never echo the site's href attribute
    or build an arbitrary redirect from it. No credentials appear in the URL.
    """
    if type(order_id) is not str or not _ORDER_ID.fullmatch(order_id):
        return None
    return 'https://funpay.com/orders/' + order_id + '/'


def order_button(order_id):
    """One safe Telegram URL button, or None for malformed order IDs."""
    url = order_url(order_id)
    if url is None:
        return None
    return {'text': '🔗 Открыть заказ #' + order_id[:32], 'url': url}

"""Local-only plugins. External code must be manually reviewed before enabling."""
from dataclasses import dataclass
import importlib.util
import json
import os
import re
from pathlib import Path
import tempfile

_CMD = re.compile(r"^[a-z][a-z0-9_]{0,30}$")
_RESERVED = {"start", "help", "menu", "status", "plugins", "ping", "funpay"}


@dataclass(frozen=True)
class PluginInfo:
    name: str
    version: str
    command: str
    description: str


class PluginStateError(Exception):
    """Refuse to load plugins when persisted enable/disable state is unsafe."""


class PluginManager:
    def __init__(self, base_dir: Path, extra_dir: Path | None = None,
                 state_file: Path | None = None, *, allow_external=False):
        self.base_dirs = [Path(base_dir)] + ([Path(extra_dir)] if extra_dir is not None else [])
        self.state_file = Path(state_file) if state_file is not None else None
        self.allow_external = allow_external
        self.state = self._read_state()
        self.commands = {}
        self.available = {}
        self.errors = []
        self.load()

    def _read_state(self):
        if not self.state_file:
            return {}
        try:
            # A dangling symlink is not an absent state file.
            if self.state_file.is_symlink():
                raise PluginStateError("Состояние плагинов не должно быть ссылкой")
            if not self.state_file.exists():
                return {}
            if not self.state_file.is_file():
                raise PluginStateError("Некорректный путь файла состояния плагинов")
            raw = json.loads(self.state_file.read_text(encoding="utf-8"))
            if not isinstance(raw, dict) or any(
                not isinstance(k, str) or not _CMD.fullmatch(k) or type(v) is not bool
                for k, v in raw.items()
            ):
                raise PluginStateError("Повреждён файл состояния плагинов")
            return raw
        except (OSError, ValueError, UnicodeError) as exc:
            raise PluginStateError("Невозможно безопасно прочитать состояние плагинов") from None

    def set_enabled(self, command, enabled):
        if command not in self.available or type(enabled) is not bool:
            return False
        previous = self.state.copy()
        self.state[command] = enabled
        if self.state_file:
            tmp = None
            try:
                if self.state_file.is_symlink():
                    raise OSError("unsafe state file")
                self.state_file.parent.mkdir(parents=True, exist_ok=True, mode=0o700)
                with tempfile.NamedTemporaryFile(
                    mode="w", encoding="utf-8", dir=self.state_file.parent,
                    prefix=".plugin-state-", delete=False
                ) as fp:
                    tmp = Path(fp.name)
                    os.chmod(fp.name, 0o600)
                    json.dump(self.state, fp, ensure_ascii=False)
                    fp.write("\n")
                    fp.flush()
                    os.fsync(fp.fileno())
                os.replace(tmp, self.state_file)
            except OSError:
                self.state = previous
                return False
            finally:
                if tmp is not None:
                    try:
                        tmp.unlink(missing_ok=True)
                    except OSError:
                        pass
        self.load()
        return True

    def load(self):
        self.commands = {}
        self.available = {}
        self.errors = []
        for index, base in enumerate(self.base_dirs):
            # Built-in plugin directory is part of a trusted git checkout.
            # External plugins are disabled in the systemd runtime by default.
            if index > 0 and not self.allow_external:
                self.errors.append("Сторонние плагины отключены политикой безопасности")
                continue
            if not base.is_dir() or base.is_symlink():
                continue
            for folder in sorted(base.iterdir()):
                if not folder.is_dir() or folder.is_symlink():
                    continue
                manifest_file, source_file = folder / "manifest.json", folder / "plugin.py"
                if (not manifest_file.is_file() or manifest_file.is_symlink()
                        or not source_file.is_file() or source_file.is_symlink()):
                    continue
                try:
                    if manifest_file.stat().st_size > 16_384:
                        raise ValueError("Manifest too large")
                    data = json.loads(manifest_file.read_text(encoding="utf-8"))
                    if not isinstance(data, dict):
                        raise ValueError("Invalid manifest")
                    name = str(data["command"])
                    if not _CMD.fullmatch(name) or name in _RESERVED or name in self.available:
                        raise ValueError("Invalid command")
                    info = PluginInfo(
                        str(data["name"]).replace("\n", " ").replace("\r", " ")[:70],
                        str(data.get("version", "0"))[:20], name,
                        str(data.get("description", "")).replace("\n", " ")[:140]
                    )
                    self.available[name] = info
                    if not self.state.get(name, data.get("enabled") is True):
                        continue
                    spec = importlib.util.spec_from_file_location("fpr_" + folder.name, source_file)
                    if spec is None or spec.loader is None:
                        raise ValueError("Invalid plugin")
                    module = importlib.util.module_from_spec(spec)
                    spec.loader.exec_module(module)
                    handler = getattr(module, "handle", None)
                    if not callable(handler):
                        raise ValueError("Missing handle")
                    self.commands[name] = (info, handler)
                except Exception as exc:
                    self.errors.append(f"{folder.name}: {type(exc).__name__}")

    def list_plugins(self):
        return [info for info, _ in self.commands.values()]

    def list_all_plugins(self):
        return [(info, name in self.commands) for name, info in self.available.items()]

    def execute(self, command, args):
        item = self.commands.get(command)
        if not item:
            return None
        try:
            return str(item[1](args[:500]))[:3500]
        except Exception:
            return "❌ Ошибка плагина. Проверьте журнал службы."

"""Local, owner-controlled admin roles and notification preferences.

Settings are written atomically in /var/lib/funpayrus; secrets are never stored
here. A damaged settings file fails closed instead of granting default access.
"""
import json
import os
from pathlib import Path
import tempfile


class SettingsError(Exception):
    """Invalid or unsafe local settings."""


class Settings:
    MAX_ADMINS = 5

    def __init__(self, owner_id: int, filename: Path):
        if type(owner_id) is not int or owner_id <= 0:
            raise ValueError('Invalid owner id')
        self.owner_id = owner_id
        self.filename = Path(filename)
        self._admins = {}
        self._startup_alerts = set()
        self._order_alerts = set()
        self._order_monitor = False
        self._read()

    def _read(self):
        try:
            if self.filename.is_symlink():
                raise SettingsError('Файл настроек является ссылкой')
            if not self.filename.exists():
                return
            if not self.filename.is_file() or self.filename.stat().st_size > 16384:
                raise SettingsError('Некорректный файл настроек')
            raw = json.loads(self.filename.read_text(encoding='utf-8'))
            if (not isinstance(raw, dict) or not {'admins', 'startup_alerts'} <= set(raw) or
                    not set(raw) <= {'admins', 'startup_alerts', 'order_alerts', 'order_monitor'}):
                raise SettingsError('Некорректная структура настроек')
            admins = raw['admins']
            alerts = raw['startup_alerts']
            order_alerts = raw.get('order_alerts', [])
            order_monitor = raw.get('order_monitor', False)
            if type(order_monitor) is not bool:
                raise SettingsError('Некорректный режим мониторинга')
            if not isinstance(admins, dict) or len(admins) > self.MAX_ADMINS:
                raise SettingsError('Некорректные администраторы')
            parsed = {}
            for key, role in admins.items():
                if (not isinstance(key, str) or not key.isascii() or
                    not key.isdecimal() or int(key) <= 0 or str(int(key)) != key or
                    int(key) == self.owner_id or role not in ('admin', 'viewer')):
                    raise SettingsError('Некорректная роль')
                parsed[int(key)] = role
            if not isinstance(alerts, list) or len(alerts) > self.MAX_ADMINS + 1:
                raise SettingsError('Некорректные уведомления')
            parsed_alerts = set()
            for uid in alerts:
                if type(uid) is not int or uid <= 0 or uid not in parsed and uid != self.owner_id:
                    raise SettingsError('Некорректный получатель уведомлений')
                parsed_alerts.add(uid)
            if not isinstance(order_alerts, list) or len(order_alerts) > self.MAX_ADMINS + 1:
                raise SettingsError('Некорректные уведомления о заказах')
            parsed_order_alerts = set()
            for uid in order_alerts:
                if type(uid) is not int or uid <= 0 or uid not in parsed and uid != self.owner_id:
                    raise SettingsError('Некорректный получатель заказов')
                parsed_order_alerts.add(uid)
            self._admins = parsed
            self._startup_alerts = parsed_alerts
            self._order_alerts = parsed_order_alerts
            self._order_monitor = order_monitor
        except (OSError, UnicodeError, ValueError, TypeError) as exc:
            raise SettingsError('Не удалось безопасно прочитать настройки') from None

    def role(self, user_id):
        if type(user_id) is not int:
            return None
        if user_id == self.owner_id:
            return 'owner'
        return self._admins.get(user_id)

    def list_admins(self):
        return [(self.owner_id, 'owner')] + sorted(self._admins.items())

    def startup_enabled(self, user_id):
        return user_id in self._startup_alerts and self.role(user_id) is not None

    def alert_recipients(self):
        return sorted(self._startup_alerts)

    def order_alert_enabled(self, user_id):
        return user_id in self._order_alerts and self.role(user_id) in ('owner', 'admin')

    def order_alert_recipients(self):
        return sorted(uid for uid in self._order_alerts if self.role(uid) in ('owner', 'admin'))

    def monitor_enabled(self):
        return self._order_monitor

    def set_monitor(self, enabled):
        if type(enabled) is not bool:
            return False
        self._save(dict(self._admins), set(self._startup_alerts), monitor=enabled)
        return True

    def toggle_order_alert(self, user_id):
        if self.role(user_id) not in ('owner', 'admin'):
            return False
        alerts = set(self._order_alerts)
        if user_id in alerts:
            alerts.remove(user_id)
        else:
            alerts.add(user_id)
        self._save(dict(self._admins), set(self._startup_alerts), order_alerts=alerts)
        return True

    def _save(self, admins, alerts, order_alerts=None, monitor=None):
        if order_alerts is None:
            order_alerts = set(self._order_alerts)
        if monitor is None:
            monitor = self._order_monitor
        tmp = None
        try:
            if self.filename.is_symlink():
                raise OSError('unsafe symlink')
            parent = self.filename.parent
            parent.mkdir(mode=0o700, parents=True, exist_ok=True)
            data = {'admins': {str(uid): role for uid, role in sorted(admins.items())},
                    'startup_alerts': sorted(alerts),
                    'order_alerts': sorted(order_alerts),
                    'order_monitor': monitor}
            with tempfile.NamedTemporaryFile(mode='w', encoding='utf-8',
                                             dir=parent, prefix='.settings-', delete=False) as fp:
                tmp = Path(fp.name)
                os.chmod(fp.name, 0o600)
                json.dump(data, fp, ensure_ascii=False)
                fp.write('\n')
                fp.flush()
                os.fsync(fp.fileno())
            os.replace(tmp, self.filename)
            fd = os.open(parent, os.O_RDONLY)
            try:
                os.fsync(fd)
            finally:
                os.close(fd)
        except OSError:
            raise SettingsError('Не удалось сохранить настройки') from None
        finally:
            if tmp is not None:
                try:
                    tmp.unlink(missing_ok=True)
                except OSError:
                    pass
        self._admins = admins
        self._startup_alerts = alerts
        self._order_alerts = order_alerts
        self._order_monitor = monitor

    def add(self, user_id, role='viewer'):
        if (type(user_id) is not int or user_id <= 0 or user_id == self.owner_id or
            role not in ('viewer', 'admin')):
            return False
        if user_id not in self._admins and len(self._admins) >= self.MAX_ADMINS:
            return False
        new = dict(self._admins)
        new[user_id] = role
        self._save(new, set(self._startup_alerts))
        return True

    def remove(self, user_id):
        if user_id not in self._admins:
            return False
        new = dict(self._admins)
        del new[user_id]
        self._save(new, self._startup_alerts - {user_id},
                   order_alerts=self._order_alerts - {user_id})
        return True

    def toggle_startup(self, user_id):
        if self.role(user_id) is None:
            return False
        alerts = set(self._startup_alerts)
        if user_id in alerts:
            alerts.remove(user_id)
        else:
            alerts.add(user_id)
        self._save(dict(self._admins), alerts)
        return True

"""Telegram Bot API, standard library only; never leak bot token into logs."""
import json
from urllib import request, error


class TelegramError(Exception):
    pass


class TelegramAPI:
    def __init__(self, token):
        self._token = token

    def call(self, method, payload=None):
        # Only internal constant method names are intended; reject injection early.
        if method not in {"getMe", "sendMessage", "editMessageText", "answerCallbackQuery", "getUpdates"}:
            raise TelegramError("Недопустимый метод Telegram API")
        url = "https://api.telegram.org/bot" + self._token + "/" + method
        req = request.Request(
            url, data=json.dumps(payload or {}, ensure_ascii=False).encode("utf-8"),
            headers={"Content-Type": "application/json"}, method="POST"
        )
        try:
            with request.urlopen(req, timeout=40) as resp:
                result = resp.read(2 * 1024 * 1024 + 1)
                if len(result) > 2 * 1024 * 1024:
                    raise TelegramError("Слишком большой ответ Telegram API")
                data = json.loads(result.decode("utf-8"))
        except error.HTTPError as exc:
            raise TelegramError(f"Telegram API HTTP {exc.code}") from None
        except (error.URLError, TimeoutError, OSError):
            raise TelegramError("Нет соединения с Telegram API") from None
        except (UnicodeError, ValueError):
            raise TelegramError("Некорректный ответ Telegram API") from None
        if not isinstance(data, dict) or not data.get("ok"):
            raise TelegramError("Telegram API отказал в выполнении запроса")
        return data.get("result")

    def send(self, chat_id, message, reply_markup=None):
        data = {"chat_id": chat_id, "text": message[:4000]}
        if reply_markup:
            data["reply_markup"] = reply_markup
        return self.call("sendMessage", data)

    def edit(self, chat_id, message_id, message, reply_markup=None):
        data = {"chat_id": chat_id, "message_id": message_id, "text": message[:4000]}
        if reply_markup:
            data["reply_markup"] = reply_markup
        return self.call("editMessageText", data)

    def acknowledge(self, callback_id):
        return self.call("answerCallbackQuery", {"callback_query_id": callback_id})

    def updates(self, offset):
        payload = {"timeout": 25, "allowed_updates": ["message", "callback_query"]}
        if offset is not None:
            payload["offset"] = offset
        updates = self.call("getUpdates", payload)
        return updates if isinstance(updates, list) else []
