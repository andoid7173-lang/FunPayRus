"""Run Telegram admin bot with private role-aware dispatch and long polling."""
from datetime import datetime, timezone
import logging
import os
from pathlib import Path
import signal
import threading
import time

from bot.checkpoint import CheckpointError, UpdateCheckpoint
from bot.commands import reply_for, callback_for, keyboard_for_command
from bot.config import Config
from bot.plugins import PluginManager, PluginStateError
from bot.settings import Settings, SettingsError

from bot.telegram_api import TelegramAPI, TelegramError

LOG = logging.getLogger('funpayrust')
RUNNING = True
STATE_DIR = Path('/var/lib/funpayrus')


def stop(_signal, _frame):
    global RUNNING
    RUNNING = False




class PollGuard:
    """Serialize periodic and manual requests and throttle forced refreshes."""

    def __init__(self, minimum_interval=20):
        self.lock = threading.Lock()
        self.last_started = None
        self.minimum_interval = minimum_interval

    def enter(self):
        if not self.lock.acquire(blocking=False):
            return False
        now = time.monotonic()
        if self.last_started is not None and now - self.last_started < self.minimum_interval:
            self.lock.release()
            return False
        self.last_started = now
        return True

    def leave(self):
        self.lock.release()


def flush_deliveries(store, settings, api, limit=30):
    """Durably retry unsent Telegram notifications; never leak to revoked roles.

    At-least-once delivery: a process crash between Telegram success and
    local acknowledgement may produce a duplicate, but never silently
    discards the notification before Telegram accepts it.
    """
    sent = 0
    try:
        entries = store.pending_deliveries(limit=limit)
    except OrderMonitorError as exc:
        LOG.warning('Очередь уведомлений: %s', exc)
        return 0
    for event_id, uid, kind, previous_status, order in entries:
        if settings.role(uid) not in ('owner', 'admin') or not settings.order_alert_enabled(uid):
            try:
                store.mark_delivered(event_id)
            except OrderMonitorError:
                LOG.warning('Не удалось сохранить отзыв подписки; повторим позднее')
                return sent
            continue
        button = order_button(order.order_id)
        markup = {'inline_keyboard': [[button]]} if button else None
        if kind == 'paid':
            title = '🛒 Новый оплаченный заказ'
        else:
            old = previous_status or 'неизвестно'
            title = f'🔄 Статус заказа изменился: {old} → {order.status}'
        try:
            api.send(uid, title + '\\n\\n' + describe_order(order), markup)
        except TelegramError:
            LOG.warning('Уведомление о заказе не доставлено; повторим позднее')
            continue
        try:
            store.mark_delivered(event_id)
        except OrderMonitorError:
            LOG.warning('Не удалось подтвердить доставку; возможен повтор после восстановления БД')
            return sent
        sent += 1
    return sent


def poll_orders_once(store, settings, api, key=None, fetcher=fetch_sales, guard=None):
    """One opt-in read-only polling cycle; sends durably queued events."""
    if not settings.monitor_enabled():
        flush_deliveries(store, settings, api)
        return 0
    if guard is not None and not guard.enter():
        flush_deliveries(store, settings, api)
        return 0
    try:
        key = os.environ.get('FUNPAY_GOLDEN_KEY') if key is None else key
        if not key:
            store.last_error = 'Golden Key отсутствует. Настрой его только на VPS.'
            return 0
        try:
            updates = store.record(
                fetcher(key, user_agent=os.getenv('FUNPAY_USER_AGENT')),
                recipients=settings.order_alert_recipients()
            )
        except OrderMonitorError as exc:
            store.last_error = str(exc)
            return 0
        return len(updates)
    finally:
        if guard is not None:
            guard.leave()
        flush_deliveries(store, settings, api)


def monitor_orders(store, settings, api, stop_event, guard=None):
    """30-second opt-in polling with backoff; alert on repeated failures."""
    failures = 0
    warned = False
    while not stop_event.is_set():
        if not settings.monitor_enabled():
            flush_deliveries(store, settings, api)
            stop_event.wait(4)
            continue
        poll_orders_once(store, settings, api, guard=guard)
        if store.last_error:
            failures += 1
            LOG.warning('Мониторинг заказов: %s', store.last_error)
            if failures >= 3 and not warned:
                try:
                    api.send(settings.owner_id, '⚠️ Мониторинг FunPay: три проверки подряд завершились ошибкой. '
                             'Открой /orders для диагностики. Покупателям ничего не отправлялось.')
                    warned = True
                except TelegramError:
                    LOG.warning('Предупреждение о сбое FunPay не доставлено')
        else:
            if warned:
                try:
                    api.send(settings.owner_id, '✅ Мониторинг FunPay снова получает данные.')
                except TelegramError:
                    pass
            failures = 0
            warned = False
        delay = 300 if store.last_error and 'частоту' in store.last_error else 30
        stop_event.wait(delay)


def main():
    logging.basicConfig(level=logging.INFO, format='%(asctime)s %(levelname)s %(message)s')
    signal.signal(signal.SIGTERM, stop)
    signal.signal(signal.SIGINT, stop)
    try:
        config = Config.load()
        checkpoint = UpdateCheckpoint(STATE_DIR / 'telegram-offset.json')
        settings = Settings(config.admin_id, STATE_DIR / 'settings.json')
        orders = OrderStore(STATE_DIR / 'orders.sqlite3')
    except ValueError as exc:
        LOG.error('Ошибка настроек: %s', exc)
        return 2
    except (CheckpointError, SettingsError, OrderMonitorError) as exc:
        LOG.error('Ошибка безопасного хранения: %s', exc)
        return 2

    allow_external = os.getenv('FUNPAYRUS_ALLOW_EXTERNAL_PLUGINS') == '1'
    try:
        plugins = PluginManager(Path(__file__).resolve().parent.parent / 'plugins',
                                STATE_DIR / 'plugins', STATE_DIR / 'plugin-state.json',
                                allow_external=allow_external)
    except PluginStateError as exc:
        LOG.error('Ошибка состояния плагинов: %s', exc)
        return 2
    if not allow_external:
        LOG.warning('Сторонние плагины выключены до ручного разрешения')
    api = TelegramAPI(config.bot_token)
    try:
        me = api.call('getMe')
    except TelegramError as exc:
        LOG.error('Не удалось запустить бота: %s', exc)
        return 1
    LOG.info('FunPay Rust стартовал (Telegram bot id %s)', me.get('id'))
    for uid in settings.alert_recipients():
        try:
            api.send(uid, '✅ FunPay Rust запущен. Служба работает.')
        except TelegramError:
            LOG.warning('Не удалось доставить уведомление об автозапуске (chat id скрыт)')
    stop_event = threading.Event()
    poll_guard = PollGuard()
    def poll_now():
        return poll_orders_once(orders, settings, api, guard=poll_guard)
    threading.Thread(target=monitor_orders, args=(orders, settings, api, stop_event, poll_guard),
                     name='funpay-orders-readonly', daemon=True).start()
    started_at = datetime.now(timezone.utc)
    while RUNNING:
        try:
            for update in api.updates(checkpoint.last_id + 1):
                uid = update.get('update_id')
                try:
                    if not checkpoint.mark(uid):
                        continue
                except CheckpointError as exc:
                    LOG.error('Блокировка обработки: %s', exc)
                    return 2
                message = update.get('message') or {}
                answer = reply_for(message, config.admin_id, plugins, started_at,
                                   settings=settings, orders=orders, poll_now=poll_now)
                if answer:
                    chat_id = message['from']['id']
                    buttons = keyboard_for_command(message.get('text', ''), plugins,
                                                   settings, chat_id, config.admin_id, orders)
                    api.send(chat_id, answer, buttons)
                callback = update.get('callback_query') or {}
                if callback:
                    response = callback_for(callback, config.admin_id, plugins, started_at,
                                            settings=settings, orders=orders, poll_now=poll_now)
                    if response:
                        text, buttons = response
                        chat_id = callback['from']['id']
                        try:
                            api.edit(chat_id, callback['message']['message_id'], text, buttons)
                        except (KeyError, TelegramError):
                            api.send(chat_id, text, buttons)
                    if isinstance(callback.get('id'), str):
                        api.acknowledge(callback['id'])
        except TelegramError as exc:
            LOG.warning('Telegram API: %s; повтор через 5 секунд', exc)
            time.sleep(5)
        except Exception as exc:
            LOG.error('Ошибка обработки события: %s', type(exc).__name__)
            time.sleep(2)
    stop_event.set()
    LOG.info('FunPay Rust остановлен')
    return 0


if __name__ == '__main__':
    raise SystemExit(main())

"""Crash-safe monotonic Telegram update checkpoint.

Persist *before* handling an update: repeated actions won't be run after a crash.
This is at-most-once dispatch, so a crash between the checkpoint and handling
may lose an update. It is not an exactly-once delivery guarantee.
"""
import json
import os
from pathlib import Path
import tempfile


class CheckpointError(Exception):
    """Checkpoint cannot be read or safely persisted."""


class UpdateCheckpoint:
    def __init__(self, filename: Path):
        self.filename = Path(filename)
        self.last_id = self._read()

    def _read(self):
        try:
            if not self.filename.exists():
                return -1
            if self.filename.is_symlink():
                raise CheckpointError("Файл позиции Telegram является ссылкой")
            obj = json.loads(self.filename.read_text(encoding="utf-8"))
            value = obj.get("last_update_id") if isinstance(obj, dict) else None
            if type(value) is not int or value < -1:
                raise ValueError("bad update id")
            return value
        except (OSError, ValueError, UnicodeError) as exc:
            raise CheckpointError("Повреждён или недоступен файл позиции Telegram") from None

    def mark(self, update_id: int) -> bool:
        """Return False for duplicate/out-of-order ids, persist newer ids."""
        if type(update_id) is not int or update_id < 0:
            return False
        if update_id <= self.last_id:
            return False
        parent = self.filename.parent
        temporary = None
        try:
            if self.filename.is_symlink():
                raise OSError("unsafe checkpoint path")
            parent.mkdir(parents=True, exist_ok=True, mode=0o700)
            with tempfile.NamedTemporaryFile(
                mode="w", encoding="utf-8", dir=parent,
                prefix=".update-checkpoint-", delete=False
            ) as fp:
                temporary = Path(fp.name)
                os.chmod(fp.name, 0o600)
                json.dump({"last_update_id": update_id}, fp)
                fp.write("\n")
                fp.flush()
                os.fsync(fp.fileno())
            os.replace(temporary, self.filename)
            fd = os.open(parent, os.O_RDONLY)
            try:
                os.fsync(fd)
            finally:
                os.close(fd)
        except OSError:
            raise CheckpointError("Не удалось безопасно сохранить позицию Telegram") from None
        finally:
            if temporary is not None:
                try:
                    temporary.unlink(missing_ok=True)
                except OSError:
                    pass
        self.last_id = update_id
        return True

"""Private Telegram interface with owner/admin/viewer access boundaries."""
from datetime import datetime, timezone

_ID = re.compile(r'^[1-9][0-9]{0,18}$')


def role_for(user_id, owner_id, settings=None):
    if settings is not None:
        return settings.role(user_id)
    return 'owner' if type(user_id) is int and user_id == owner_id else None


def allowed(message, admin_id, settings=None):
    user = message.get('from') or {}
    chat = message.get('chat') or {}
    uid = user.get('id')
    return (role_for(uid, admin_id, settings) is not None and
            chat.get('type') == 'private' and chat.get('id') == uid)

def orders_menu(settings, uid, orders=None):
    rows = []
    if settings and settings.role(uid) == 'owner':
        enabled = settings.monitor_enabled()
        rows.append([{'text': ('⏹ Выключить' if enabled else '▶️ Включить') + ' мониторинг',
                      'callback_data': 'o:toggle'}])
    # Order links come only from validated IDs stored by the read-only monitor.
    if orders is not None and settings and settings.role(uid) in ('owner', 'admin'):
        try:

            for item in items[:5]:

                if button is not None:
                    rows.append([button])
        except OrderMonitorError:
            pass  # Keep the section usable if SQLite is temporarily unavailable.
    rows.append([{'text': '⚡ Проверить сейчас', 'callback_data': 'o:check'}])
    rows.append([{'text': '🔄 Обновить экран', 'callback_data': 'm:orders'}])
    rows.append([{'text': '⬅️ Назад', 'callback_data': 'm:home'}])
    return {'inline_keyboard': rows}


def orders_screen(settings, orders):
    if not settings or orders is None:
        return '📦 Хранилище заказов не настроено.'
    enabled = settings.monitor_enabled()
    result = ['📦 Заказы FunPay — тест только на чтение',
              'Мониторинг: ' + ('✅ включён' if enabled else '⏸ выключен'),
              'Первый успешный опрос сохраняет историю без уведомлений.',
              'Опрос: примерно каждые 30 секунд, только первая страница продаж.']
    if orders.last_error:
        result.append('⚠️ Последняя ошибка: ' + orders.last_error)
    try:
        count, last, items = orders.summary()
        result.append(f'Сохранено заказов: {count}')
        result.append('Последняя проверка: ' +
                      (datetime.fromtimestamp(last, timezone.utc).strftime('%Y-%m-%d %H:%M UTC')
                       if last else 'ещё не проводилась'))
        for order in items[:5]:
            result.append(f'#{order.order_id}: {order.status} — {order.description[:65]}')
    except OrderMonitorError:
        result.append('⚠️ Не удалось прочитать историю заказов.')
    result.append('Для владельца: /orders on или /orders off.\n'
                  'Уведомления включаются отдельно в 🔔 Уведомления.\n'
                  'Никаких сообщений покупателям и автовыдачи.')
    return '\n'.join(result)[:3800]

def _add_admin(args, settings):
    parts = args.split()
    if len(parts) not in (1, 2) or not _ID.fullmatch(parts[0]):
        return 'Используй: /addadmin <Telegram ID> [viewer|admin]'
    role = parts[1] if len(parts) == 2 else 'viewer'
    if role not in ('admin', 'viewer'):
        return 'Роль должна быть viewer или admin.'
    try:
        ok = settings.add(int(parts[0]), role)
    except SettingsError:
        return '❌ Не удалось безопасно сохранить настройки.'
    return ('✅ Доступ настроен. Попроси пользователя открыть бота и нажать /start.'
            if ok else '❌ Нельзя добавить пользователя. Проверь ID и лимит (5).')


def _remove_admin(args, settings):
    if not _ID.fullmatch(args.strip()):
        return 'Используй: /removeadmin <Telegram ID>'
    try:
        ok = settings.remove(int(args.strip()))
    except SettingsError:
        return '❌ Не удалось безопасно сохранить настройки.'
    return '✅ Доступ отозван.' if ok else '❌ Пользователь не найден или владелец.'


def reply_for(message, admin_id, plugins, started_at,
              funpay_status=status_from_environment, settings=None, orders=None, poll_now=None):
    if not allowed(message, admin_id, settings):
        return None
    text = message.get('text')
    if not isinstance(text, str) or not text.strip().startswith('/'):
        return None
    parts = text.strip().split(maxsplit=1)
    command = parts[0][1:].split('@')[0].lower()
    args = parts[1] if len(parts) == 2 else ''
    uid = message['from']['id']
    role = role_for(uid, admin_id, settings)

    if command in ('start', 'help', 'menu'):
        return (f'🐱 FunPay Rust v{__version__}\n\n'
                f'Твоя роль: {role}\n\n'
                '/status — состояние\n/diagnostics — диагностика VPS\n'
                '/notifications — центр уведомлений\n/demo — демо-режим\n'
                '/setup — мастер настройки\n/funpay — тест соединения\n'
                + ('/orders — мониторинг продаж (только чтение)\n' if role in ('owner', 'admin') else '')
                + ('/admins — управление доступом\n/plugins — плагины\n' if role == 'owner' else '')
                + '\n⚠️ Автовыдача и отправка сообщений покупателям отключены.')
    if command == 'status':
        seconds = max(0, int((datetime.now(timezone.utc) - started_at).total_seconds()))
        return (f'✅ FunPay Rust v{__version__} работает\n'
                f'⏱ Время работы: {seconds // 60} мин.\n'
                f'👤 Роль: {role}\n'
                f'🧩 Активных плагинов: {len(plugins.commands)}')
    if command == 'ping':
        return '🏓 Pong! Бот на связи.'
    if command == 'diagnostics':
        return report(plugins, settings)
    if command == 'setup':
        return setup_summary(settings)
    if command == 'demo':
        return demo_screen()
    if command == 'notifications':
        return notifications_text(settings, uid)
    if command == 'funpay':
        return funpay_status() if role in ('owner', 'admin') else '🔒 Проверка FunPay доступна администраторам.'
    if command == 'testalert':
        if role not in ('owner', 'admin'):
            return '🔒 Тест доступен только администраторам.'
        return ('🧪 ТЕСТ УВЕДОМЛЕНИЯ FunPay Rust\\n\\n'
                'Это проверка доставки в Telegram. Реальной покупки нет, '
                'к FunPay запрос не отправлялся.')
    if command == 'findorder':
        if role not in ('owner', 'admin'):
            return '🔒 Поиск заказов доступен только администраторам.'
        if not 1 <= len(args.strip()) <= 80:
            return 'Используй: /findorder <номер заказа, покупатель или товар>'
        try:
            matches = orders.find(args.strip()) if orders else []
        except OrderMonitorError:
            return '⚠️ Поиск временно недоступен.'
        if not matches:
            return '🔎 Совпадений в сохранённой истории нет. Это не поиск на сервере FunPay.'
        return ('🔎 Результаты поиска (локальная история):\\n\\n'
                + '\\n\\n'.join(
                    f'#{x.order_id}: {x.status} — {x.description[:70]}'
                    for x in matches))
    if command == 'orders':
        if role not in ('owner', 'admin'):
            return '🔒 Заказы доступны только администраторам.'
        if args.strip() == 'now':
            if settings is None or not settings.monitor_enabled():
                return '⚠️ Сначала включи мониторинг: /orders on'
            if poll_now is None:
                return '⚠️ Ручная проверка недоступна в этом режиме.'
            poll_now()
        elif args.strip() in ('on', 'off'):
            if role != 'owner' or settings is None:
                return '🔒 Только владелец включает мониторинг.'
            try:
                settings.set_monitor(args.strip() == 'on')
            except SettingsError:
                return '❌ Не удалось сохранить режим мониторинга.'
        elif args.strip():
            return 'Используй: /orders [on|off|now]'
        return orders_screen(settings, orders)
    if command == 'admins':
        return admins_text(settings) if role == 'owner' else '🔒 Только владелец управляет доступом.'
    if command == 'addadmin':
        return _add_admin(args, settings) if role == 'owner' and settings else '🔒 Только владелец управляет доступом.'
    if command == 'removeadmin':
        return _remove_admin(args, settings) if role == 'owner' and settings else '🔒 Только владелец управляет доступом.'
    if command == 'plugins':
        return plugins_text(plugins) if role == 'owner' else '🔒 Доступ к плагинам ограничен владельцем.'
    return plugins.execute(command, args) if role == 'owner' else None


def keyboard_for_command(text, plugins, settings=None, user_id=None, owner_id=None, orders=None):
    command = text.strip().split(maxsplit=1)[0].split('@')[0].lower() if text and text.strip() else ''
    if command == '/plugins' and role_for(user_id, owner_id, settings) == 'owner':
        return plugin_menu(plugins)
    if command == '/notifications':
        return alerts_menu(settings, user_id)
    if command == '/orders':
        return orders_menu(settings, user_id, orders)
    if command == '/findorder' and orders and settings and settings.role(user_id) in ('owner', 'admin'):
        return {'inline_keyboard': [[b] for b in (
            order_button(o.order_id) for o in orders.find(text.partition(' ')[2])
        ) if b] + [[{'text': '⬅️ Назад', 'callback_data': 'm:orders'}]]}
    return menu(role_for(user_id, owner_id, settings))


def callback_for(callback, admin_id, plugins, started_at,
                 funpay_status=status_from_environment, settings=None, orders=None, poll_now=None):
    check = {'from': callback.get('from'), 'chat': (callback.get('message') or {}).get('chat')}
    if not allowed(check, admin_id, settings):
        return None
    uid = check['from']['id']
    role = role_for(uid, admin_id, settings)
    data = callback.get('data')
    sections = {'m:home': '/start', 'm:help': '/start', 'm:status': '/status',
                'm:ping': '/ping', 'm:funpay': '/funpay',
                'm:plugins': '/plugins', 'm:diagnostics': '/diagnostics',
                'm:setup': '/setup', 'm:demo': '/demo',
                'm:notifications': '/notifications', 'm:admins': '/admins',
                'm:orders': '/orders'}
    if data in sections:
        command = sections[data]
        output = reply_for({**check, 'text': command}, admin_id, plugins, started_at,
                           funpay_status=funpay_status, settings=settings, orders=orders, poll_now=poll_now)
        return output, keyboard_for_command(command, plugins, settings, uid, admin_id, orders)
    if data == 'o:check':
        if role not in ('owner', 'admin') or settings is None:
            return '🔒 Проверка доступна администраторам.', menu(role)
        if not settings.monitor_enabled():
            return '⚠️ Сначала включи мониторинг.', orders_menu(settings, uid, orders)
        if poll_now is not None:
            poll_now()
        return orders_screen(settings, orders), orders_menu(settings, uid, orders)
    if data == 'o:toggle':
        if role != 'owner' or settings is None:
            return '🔒 Только владелец включает мониторинг.', menu(role)
        try:
            settings.set_monitor(not settings.monitor_enabled())
        except SettingsError:
            return '❌ Не удалось сохранить режим мониторинга.', orders_menu(settings, uid, orders)
        return orders_screen(settings, orders), orders_menu(settings, uid, orders)
    if data == 'a:orders':
        if role not in ('owner', 'admin') or settings is None:
            return '🔒 Уведомления о заказах недоступны.', menu(role)
        try:
            settings.toggle_order_alert(uid)
        except SettingsError:
            return '❌ Не удалось сохранить уведомления.', alerts_menu(settings, uid)
        return notifications_text(settings, uid), alerts_menu(settings, uid)
    if data == 'a:startup':
        try:
            if not settings or not settings.toggle_startup(uid):
                return '❌ Не удалось изменить настройки.', alerts_menu(settings, uid)
        except SettingsError:
            return '❌ Не удалось сохранить настройки.', alerts_menu(settings, uid)
        return notifications_text(settings, uid), alerts_menu(settings, uid)
    if isinstance(data, str) and data.startswith('p:toggle:'):
        if role != 'owner':
            return '🔒 Только владелец управляет плагинами.', menu(role)
        command = data[len('p:toggle:'):]
        if command not in plugins.available:
            return '❌ Плагин не найден.', plugin_menu(plugins)
        if not plugins.set_enabled(command, command not in plugins.commands):
            return '❌ Не удалось сохранить состояние плагина.', plugin_menu(plugins)
        return plugins_text(plugins), plugin_menu(plugins)
    return None

"""Only reads variables provided by the systemd EnvironmentFile."""
import os
import re
from dataclasses import dataclass

_TOKEN = re.compile(r"^[0-9]{5,15}:[A-Za-z0-9_-]{30,}$")

@dataclass(frozen=True)
class Config:
    bot_token: str
    admin_id: int

    @classmethod
    def load(cls, env=None):
        env = os.environ if env is None else env
        token = str(env.get("BOT_TOKEN", "")).strip()
        admin = str(env.get("ADMIN_ID", "")).strip()
        if not _TOKEN.fullmatch(token):
            raise ValueError("Неверный формат BOT_TOKEN; настройте токен через установщик.")
        if not admin.isascii() or not admin.isdecimal() or int(admin) <= 0:
            raise ValueError("ADMIN_ID должен быть положительным числом.")
        return cls(bot_token=token, admin_id=int(admin))
